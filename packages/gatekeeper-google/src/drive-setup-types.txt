// Types for the Google Drive Folder gatekeeper.
// Exposed to the AI agent as part of the Google gatekeeper's TypeScript type catalog.

/** A file or folder entry returned by listFiles(). */
export type DriveFileEntry = {
  id: string;
  name: string;
  /** MIME type — e.g. "application/vnd.google-apps.folder", "application/vnd.google-apps.document",
   *  "application/vnd.google-apps.spreadsheet". */
  mimeType: string;
  /** Web URL to open the file in a browser (present for Google Workspace files). */
  url?: string;
  modifiedTime?: string;
  /** File size in bytes as a string. Absent for Google Workspace files (Docs, Sheets, etc.). */
  size?: string;
};

/** Metadata for a Google Drive folder. */
export type DriveFolderInfo = {
  id: string;
  name: string;
  /** Web URL to open the folder in Google Drive. */
  url: string;
};

/**
 * Session for the Google Drive Folder connection.
 *
 * PURPOSE:
 *   Query the contents of a connected Google Drive folder and create new subfolders,
 *   Google Docs, and Google Sheets inside it — including inside nested subfolders —
 *   all scoped to the bound folder. Raw Drive IDs are never accepted from agent code.
 *
 * IMPORTANT CONSTRAINTS:
 *   ✓ listFiles(), listFilesAtPath(), and getFolder() are read-only observations
 *     (no approval needed).
 *   ✓ createFolder(), createDoc(), createSheet(), scaffoldFolders(),
 *     createDocAtPath(), createSheetAtPath(), trashFile(), and moveFileToPath()
 *     require user approval before executing.
 *   ✓ All create/scaffold operations are idempotent: existing items with the same name
 *     are reused rather than duplicated. Safe to re-run on repeat calls.
 *   ✓ scaffoldFolders() creates nested subfolders under a relative path inside the bound
 *     folder — no new Drive binding is needed for each student/aluno.
 *   ✓ createDocAtPath() and createSheetAtPath() create a Doc or Sheet inside a nested
 *     subfolder path, also without requiring a new Drive binding per aluno.
 *   ✓ listFilesAtPath() lists files inside a nested path under the bound folder without
 *     requiring a separate binding per subfolder.
 *   ✓ trashFile() is a SOFT DELETE — file moves to Drive Trash, NOT permanently deleted.
 *     Always call listFiles()/listFilesAtPath() first to confirm the correct fileId.
 *   ✓ moveFileToPath() moves a file into a relative path under the bound folder.
 *     Intermediate folders are created if needed.
 *   ✗ Does NOT permanently delete files (use the Drive Trash UI for that).
 *   ✗ Does NOT send email (use the Gmail binding for that).
 *   ✗ Does NOT expose raw Drive folder IDs for creation/scaffolding; use relative names/paths.
 *   ✗ Does NOT support file uploads, sharing/permissions changes, Drive-wide search,
 *     file export, or batch file operations — those remain outside this session's scope.
 *
 * WORKFLOW for creating items:
 *   1. Call any create/scaffold method to queue the creation for approval.
 *   2. User approves the action in the Workshop UI.
 *   3. Call listFiles() or listFilesAtPath() to verify the new items.
 *
 * NESTED FOLDER SCAFFOLDING (scaffoldFolders):
 *   One "Alunos" Drive binding can scaffold per-student subfolders without a new binding
 *   per student. Example — create Documentos, Fotos, Relatorios Semanais inside each
 *   aluno's folder:
 *
 *     await drive.scaffoldFolders("Ana Silva", ["Documentos", "Fotos", "Relatorios Semanais"]);
 *     await drive.scaffoldFolders("Bruno Costa", ["Documentos", "Fotos", "Relatorios Semanais"]);
 *
 *   Or with a deeper path (creates intermediate folders automatically):
 *
 *     await drive.scaffoldFolders(["Turma A", "Ana Silva"], ["Documentos", "Fotos"]);
 *
 * NESTED DOC / SHEET CREATION (createDocAtPath / createSheetAtPath):
 *   After scaffolding folders you can place a Doc or Sheet inside any nested path without
 *   a new Drive binding. Uses the same parentPath convention as scaffoldFolders.
 *
 *   Example — create a weekly-report Doc inside each aluno's Relatorios Semanais folder:
 *
 *     await drive.createDocAtPath(
 *       ["Ana Silva", "Relatorios Semanais"],
 *       "Semana 2026-09-21",
 *       "Relatório semanal de Ana Silva\n\nAtividades:\n- ...",
 *     );
 *
 *   Example — create a tracking Sheet inside a nested folder:
 *
 *     await drive.createSheetAtPath(
 *       "Ana Silva",
 *       "Frequencia",
 *       [["Data", "Presente", "Observacoes"]],
 *     );
 *
 * CONNECTING A SPECIFIC FOLDER:
 *   Connect with a folder URL such as https://drive.google.com/drive/folders/<folderId>.
 *   Use folderId "root" or the My Drive URL to access the root of your Drive.
 */
export interface GoogleDriveFolderSession {
  /**
   * List all non-trashed files and folders directly inside the connected folder.
   * Returns an observation (no approval required).
   * Use this to answer "what is in my folder?" or to discover file IDs for further actions.
   */
  listFiles(): Promise<DriveFileEntry[]>;

  /**
   * Get name and URL metadata for the connected folder itself.
   * Returns an observation (no approval required).
   */
  getFolder(): Promise<DriveFolderInfo>;

  /**
   * Submit a request to create a subfolder directly inside the connected folder.
   * If a folder with the same name already exists, it will be reused after approval.
   * The action requires user approval before it is executed.
   * Call listFiles() after approval to see the new folder.
   *
   * To create a subfolder inside a nested path, use scaffoldFolders() instead.
   */
  createFolder(name: string): Promise<void>;

  /**
   * Submit a request to create a Google Doc directly inside the connected folder.
   * Optionally inserts `initialText` as the document's starting content.
   * If a Doc with the same name already exists, it will be reused after approval.
   * The action requires user approval before it is executed.
   * Call listFiles() after approval to see the new document.
   *
   * To create a Doc inside a nested subfolder, use createDocAtPath() instead.
   */
  createDoc(name: string, initialText?: string): Promise<void>;

  /**
   * Submit a request to create a Google Sheet directly inside the connected folder.
   * Optionally appends `initialRows` as the spreadsheet's starting data.
   * If a Sheet with the same name already exists, it will be reused after approval.
   * The action requires user approval before it is executed.
   * Call listFiles() after approval to see the new spreadsheet.
   *
   * To create a Sheet inside a nested subfolder, use createSheetAtPath() instead.
   */
  createSheet(name: string, initialRows?: string[][]): Promise<void>;

  /**
   * Submit a request to create one or more subfolders under a relative path inside the
   * bound folder. All path resolution starts from the bound folder — no raw Drive IDs.
   *
   * @param parentPath  A single folder name (string) or an ordered list of names (string[])
   *                    forming a relative path under the bound folder. Each intermediate
   *                    folder is created if it does not exist. Pass [] or "" to create
   *                    directly inside the bound folder.
   * @param folderNames Names of the subfolders to create under the resolved parentPath.
   *
   * Segment rules (applied to every path segment and folder name):
   *   - Must be non-empty and at most 255 characters.
   *   - Must not contain `/`, `\`, null bytes, or control characters (ASCII 0–31).
   *   - At most 10 path segments, at most 50 folder names per call.
   *
   * Idempotent: existing folders with matching names are reused, not duplicated.
   * Requires user approval before any folders are created.
   *
   * Examples:
   *   // Create Documentos, Fotos, Relatorios Semanais inside "Ana Silva" (under bound folder)
   *   await drive.scaffoldFolders("Ana Silva", ["Documentos", "Fotos", "Relatorios Semanais"]);
   *
   *   // Create the same under Turma A / Ana Silva (two-level path, created if needed)
   *   await drive.scaffoldFolders(["Turma A", "Ana Silva"], ["Documentos", "Fotos"]);
   *
   *   // Create directly inside the bound folder
   *   await drive.scaffoldFolders([], ["Documentos", "Fotos"]);
   */
  scaffoldFolders(parentPath: string | string[], folderNames: string[]): Promise<void>;

  /**
   * Submit a request to create a Google Doc under a relative path inside the bound folder.
   * All path resolution starts from the bound folder — no raw Drive IDs.
   * Intermediate folders in `parentPath` are created idempotently if absent.
   *
   * @param parentPath  A single folder name (string) or an ordered list of names (string[])
   *                    forming a relative path under the bound folder. Each intermediate
   *                    folder is created if it does not exist. Pass [] or "" to create
   *                    directly inside the bound folder (same as createDoc()).
   * @param name        Name for the new Google Doc. Must follow the segment rules below.
   * @param initialText Optional text to insert as the document's starting content.
   *
   * Segment rules (applied to every path segment and the doc name):
   *   - Must be non-empty and at most 255 characters.
   *   - Must not contain `/`, `\`, null bytes, or control characters (ASCII 0–31).
   *   - At most 10 path segments.
   *
   * Idempotent: if a Doc with the same name already exists under the resolved path, it is
   * reused rather than duplicated.
   * Requires user approval before the Doc is created.
   *
   * Examples:
   *   // Create a weekly-report Doc inside Ana Silva's Relatorios Semanais folder
   *   await drive.createDocAtPath(
   *     ["Ana Silva", "Relatorios Semanais"],
   *     "Semana 2026-09-21",
   *     "Relatório semanal de Ana Silva\n\nAtividades:\n- ...",
   *   );
   *
   *   // Create directly inside the bound folder (equivalent to createDoc)
   *   await drive.createDocAtPath([], "Ata de Reuniao");
   */
  createDocAtPath(
    parentPath: string | string[],
    name: string,
    initialText?: string,
  ): Promise<void>;

  /**
   * Submit a request to create a Google Sheet under a relative path inside the bound folder.
   * All path resolution starts from the bound folder — no raw Drive IDs.
   * Intermediate folders in `parentPath` are created idempotently if absent.
   *
   * @param parentPath  A single folder name (string) or an ordered list of names (string[])
   *                    forming a relative path under the bound folder. Each intermediate
   *                    folder is created if it does not exist. Pass [] or "" to create
   *                    directly inside the bound folder (same as createSheet()).
   * @param name        Name for the new Google Sheet. Must follow the segment rules below.
   * @param initialRows Optional 2-D array of strings to write as the sheet's starting rows.
   *
   * Segment rules (applied to every path segment and the sheet name):
   *   - Must be non-empty and at most 255 characters.
   *   - Must not contain `/`, `\`, null bytes, or control characters (ASCII 0–31).
   *   - At most 10 path segments.
   *
   * Idempotent: if a Sheet with the same name already exists under the resolved path, it is
   * reused rather than duplicated.
   * Requires user approval before the Sheet is created.
   *
   * Examples:
   *   // Create a frequency-tracking Sheet inside Ana Silva's folder
   *   await drive.createSheetAtPath(
   *     "Ana Silva",
   *     "Frequencia",
   *     [["Data", "Presente", "Observacoes"]],
   *   );
   *
   *   // Create directly inside the bound folder (equivalent to createSheet)
   *   await drive.createSheetAtPath([], "Dados Gerais");
   */
  createSheetAtPath(
    parentPath: string | string[],
    name: string,
    initialRows?: string[][],
  ): Promise<void>;

  /**
   * List files and folders inside a nested path under the bound folder.
   * No approval required — this is a read-only observation.
   * Path segments are resolved read-only (folders are NOT created if absent).
   * Returns an empty array if the path does not exist.
   *
   * @param parentPath  A single folder name (string) or ordered list of names (string[])
   *                    forming a relative path under the bound folder. Pass [] or "" to
   *                    list the bound folder itself (equivalent to listFiles()).
   * @param options.name          Optional exact-name filter.
   * @param options.mimeType      Optional MIME type filter. Common values:
   *                                "application/vnd.google-apps.folder"
   *                                "application/vnd.google-apps.document"
   *                                "application/vnd.google-apps.spreadsheet"
   * @param options.includeTrashed  Include trashed items (default false).
   * @param options.pageSize        Max results, 1–1000 (default 200).
   *
   * Examples:
   *   // List files directly inside "Ana Silva" (under bound folder)
   *   const files = await drive.listFilesAtPath("Ana Silva");
   *
   *   // List only Google Docs inside a nested path
   *   const docs = await drive.listFilesAtPath(
   *     ["Ana Silva", "Relatorios Semanais"],
   *     { mimeType: "application/vnd.google-apps.document" },
   *   );
   *
   *   // List up to 50 items including trashed ones
   *   const all = await drive.listFilesAtPath("Arquivados", { includeTrashed: true, pageSize: 50 });
   */
  listFilesAtPath(
    parentPath: string | string[],
    options?: {
      name?: string;
      mimeType?: string;
      includeTrashed?: boolean;
      pageSize?: number;
    },
  ): Promise<DriveFileEntry[]>;

  /**
   * Submit a request to soft-delete a file by moving it to Google Drive Trash.
   * This is NOT a permanent deletion — the file can be restored from the Trash at any time.
   * Requires user approval before executing.
   *
   * IMPORTANT: The `fileId` must come from a previous listFiles() or listFilesAtPath()
   * call to ensure you are trashing the intended file. Confirm the file name in the
   * approval dialog before approving.
   *
   * @param fileId  Drive file ID returned by listFiles() or listFilesAtPath().
   *                Must be a non-empty string, ≤ 255 characters, no whitespace.
   *
   * Throws if:
   *   - The file does not exist or is not accessible.
   *   - The file is already in the trash.
   *
   * Examples:
   *   // Find a file, then trash it
   *   const files = await drive.listFilesAtPath("Ana Silva");
   *   const report = files.find(f => f.name === "Relatorio Antigo");
   *   if (report) await drive.trashFile(report.id);
   */
  trashFile(fileId: string): Promise<void>;

  /**
   * Submit a request to move a file into a target folder resolved from a relative path
   * under the bound folder. All path resolution starts from the bound folder — raw Drive
   * IDs are never used for the target.
   * Intermediate folders in `targetPath` are created idempotently if absent (same as
   * scaffoldFolders). The file is removed from its current parent(s).
   * Requires user approval before executing.
   *
   * @param fileId      Drive file ID to move. Obtain from listFiles() or listFilesAtPath().
   * @param targetPath  Relative path under the bound folder to move the file into.
   *                    A single folder name (string) or ordered list of names (string[]).
   *                    Pass [] or "" to move into the bound folder itself.
   *
   * Segment rules (applied to every path segment):
   *   - Must be non-empty and at most 255 characters.
   *   - Must not contain `/`, `\`, null bytes, or control characters (ASCII 0–31).
   *   - At most 10 path segments.
   *
   * Examples:
   *   // Move a file into "Ana Silva / Arquivados" (creates Arquivados if needed)
   *   await drive.moveFileToPath(file.id, ["Ana Silva", "Arquivados"]);
   *
   *   // Move a file into a single subfolder directly inside the bound folder
   *   await drive.moveFileToPath(file.id, "Relatorios");
   *
   *   // Move a file into the bound folder itself
   *   await drive.moveFileToPath(file.id, []);
   */
  moveFileToPath(fileId: string, targetPath: string | string[]): Promise<void>;
}
